  244  *&      Form  FCODE_INSERT_ROW                                         *
  245  *&---------------------------------------------------------------------*
  246  FORM fcode_insert_row
  247                USING    p_tc_name           TYPE dynfnam
  248                         p_table_name             .
  249
  250  *&SPWIZARD: BEGIN OF LOCAL DATA----------------------------------------*
  251    DATA l_lines_name       LIKE feld-name.
  252    DATA l_selline          LIKE sy-stepl.
  253    DATA l_lastline         TYPE i.
  254    DATA l_line             TYPE i.
  255    DATA l_table_name       LIKE feld-name.
  256    FIELD-SYMBOLS <tc>                 TYPE cxtab_control.
  257    FIELD-SYMBOLS <table>              TYPE STANDARD TABLE.
  258    FIELD-SYMBOLS <lines>              TYPE i.
  259  *&SPWIZARD: END OF LOCAL DATA------------------------------------------*
  260
  261    ASSIGN (p_tc_name) TO <tc>.
  262
  263  *&SPWIZARD: get the table, which belongs to the tc                     *
  264    CONCATENATE p_table_name '[]' INTO l_table_name. "table body
  265    ASSIGN (l_table_name) TO <table>.                "not headerline
  266
  267  *&SPWIZARD: get looplines of TableControl                              *
  268    CONCATENATE 'G_' p_tc_name '_LINES' INTO l_lines_name.
  269    ASSIGN (l_lines_name) TO <lines>.
  270
  271  *&SPWIZARD: get current line                                           *
  272    GET CURSOR LINE l_selline.
  273    IF sy-subrc <> 0.                   " append line to table
  274      l_selline = <tc>-lines + 1.
  275  *&SPWIZARD: set top line                                               *
  276      IF l_selline > <lines>.
  277        <tc>-top_line = l_selline - <lines> + 1 .
  278      ELSE.
  279        <tc>-top_line = 1.
  280      ENDIF.
  281    ELSE.                               " insert line into table
  282      l_selline = <tc>-top_line + l_selline - 1.
  283      l_lastline = <tc>-top_line + <lines> - 1.
  284    ENDIF.
  285  *&SPWIZARD: set new cursor line                                        *
  286    l_line = l_selline - <tc>-top_line + 1.
  287
  288  *&SPWIZARD: insert initial line                                        *
  289    INSERT INITIAL LINE INTO <table> INDEX l_selline.
  290    <tc>-lines = <tc>-lines + 1.
  291  *&SPWIZARD: set cursor                                                 *
  292    SET CURSOR LINE l_line.
  293
  294  ENDFORM.                              " FCODE_INSERT_ROW
