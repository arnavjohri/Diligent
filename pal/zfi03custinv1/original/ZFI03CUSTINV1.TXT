  131  FORM hbank_f4                                         "Added by Paul 4/1
       2/2023
  132    CHANGING
  133    cv_hbkid TYPE hbkid.
  134
  135    DATA:
  136      lt_hbank   TYPE STANDARD TABLE OF ty_hbank,
  137      lt_rehbank TYPE STANDARD TABLE OF ddshretval.
  138
  139    SELECT t1~bukrs
  140         t1~hbkid
  141         t2~banks
  142         t2~bankl
  143         t3~banka
  144         t3~ort01
  145    FROM t012k AS t1
  146    INNER JOIN t012 AS t2
  147    ON t1~bukrs = t2~bukrs
  148    AND t1~hbkid = t2~hbkid
  149    INNER JOIN bnka AS t3
  150    ON t2~banks = t3~banks
  151    AND t2~bankl = t3~bankl
  152    INTO TABLE lt_hbank
  153    WHERE t1~bukrs EQ p_rbukrs.
  154
  155    IF sy-subrc NE 0.
  156      RETURN.
  157    ENDIF.
  158
  159    CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
  160      EXPORTING
  161        retfield        = 'HBKID'    " Name of field in VALUE_TAB
  162        value_org       = 'S'        " Value return: C: cell by cell, S: s
       tructured
  163      TABLES
  164        value_tab       = lt_hbank  " Table of values: entries cell by cel
       l
  165        return_tab      = lt_rehbank  " Return the selected value
  166      EXCEPTIONS
  167        parameter_error = 1          " Incorrect parameter
  168        no_values_found = 2          " No values found
  169        OTHERS          = 3.
  170
  171    IF sy-subrc NE 0.
  172      RETURN.
  173    ENDIF.
  174
  175    READ TABLE lt_rehbank INTO DATA(ls_rehbank) INDEX 1.
  176    cv_hbkid = ls_rehbank-fieldval.
